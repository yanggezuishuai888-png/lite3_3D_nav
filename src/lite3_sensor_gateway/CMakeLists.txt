cmake_minimum_required(VERSION 3.10)
project(lite3_sensor_gateway)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(ZLIB REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
find_package(Threads REQUIRED)
set(ros_deps rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs diagnostic_msgs tf2_msgs tf2_ros)
add_library(gateway_protocol STATIC src/protocol.cpp src/time_sync.cpp)
target_include_directories(gateway_protocol PUBLIC include)
target_link_libraries(gateway_protocol PUBLIC ZLIB::ZLIB)
add_library(gateway_codecs STATIC src/codecs.cpp)
target_include_directories(gateway_codecs PUBLIC include)
target_include_directories(gateway_codecs PUBLIC ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(gateway_codecs ${ros_deps})
target_link_libraries(gateway_codecs gateway_protocol ${OpenCV_LIBRARIES})
foreach(node sender receiver)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} ${ros_deps})
  target_link_libraries(${node} gateway_codecs Threads::Threads)
endforeach()
foreach(node command_sender command_receiver)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} ${ros_deps})
  target_link_libraries(${node} gateway_codecs)
endforeach()
add_executable(image_rate_probe test/image_rate_probe.cpp)
target_include_directories(image_rate_probe PRIVATE include)
ament_target_dependencies(image_rate_probe rclcpp sensor_msgs)
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(protocol_test test/protocol_test.cpp)
  target_link_libraries(protocol_test gateway_protocol)
  add_test(NAME protocol_test COMMAND protocol_test)
  add_executable(time_sync_test test/time_sync_test.cpp)
  target_link_libraries(time_sync_test gateway_protocol)
  add_test(NAME time_sync_test COMMAND time_sync_test)
  add_executable(codec_test test/codec_test.cpp)
  target_link_libraries(codec_test gateway_codecs)
  ament_target_dependencies(codec_test ${ros_deps})
  add_test(NAME codec_test COMMAND codec_test)
  add_executable(integration_fixture test/integration_fixture.cpp)
  target_link_libraries(integration_fixture gateway_codecs)
  ament_target_dependencies(integration_fixture ${ros_deps})
  add_test(NAME gateway_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/run_integration.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(gateway_integration PROPERTIES TIMEOUT 40)
  add_test(NAME clock_gate_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/run_integration.py ${CMAKE_CURRENT_BINARY_DIR} blocked)
  set_tests_properties(clock_gate_integration PROPERTIES TIMEOUT 25 RUN_SERIAL TRUE)
  set_tests_properties(gateway_integration PROPERTIES RUN_SERIAL TRUE)
endif()
install(TARGETS sender receiver command_sender command_receiver DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
