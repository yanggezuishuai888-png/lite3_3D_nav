cmake_minimum_required(VERSION 3.10)
project(lite3_sensor_gateway)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
set(ros_deps rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs diagnostic_msgs)
add_library(gateway_protocol STATIC src/protocol.cpp)
target_include_directories(gateway_protocol PUBLIC include)
target_link_libraries(gateway_protocol PUBLIC ZLIB::ZLIB)
add_library(gateway_codecs STATIC src/codecs.cpp)
target_include_directories(gateway_codecs PUBLIC include)
ament_target_dependencies(gateway_codecs ${ros_deps})
target_link_libraries(gateway_codecs gateway_protocol)
foreach(node sender receiver)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} ${ros_deps})
  target_link_libraries(${node} gateway_codecs Threads::Threads)
endforeach()
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(protocol_test test/protocol_test.cpp)
  target_link_libraries(protocol_test gateway_protocol)
  add_test(NAME protocol_test COMMAND protocol_test)
  add_executable(codec_test test/codec_test.cpp)
  target_link_libraries(codec_test gateway_codecs)
  ament_target_dependencies(codec_test ${ros_deps})
  add_test(NAME codec_test COMMAND codec_test)
  add_executable(integration_fixture test/integration_fixture.cpp)
  target_link_libraries(integration_fixture gateway_codecs)
  ament_target_dependencies(integration_fixture ${ros_deps})
  add_test(NAME gateway_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/run_integration.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(gateway_integration PROPERTIES TIMEOUT 40)
endif()
install(TARGETS sender receiver DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
